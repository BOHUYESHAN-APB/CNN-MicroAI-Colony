# 小数据集优化策略
class SmallDatasetColonyDetector(nn.Module):
    def __init__(self):
        super().__init__()
        self.backbone = ResNet50(pretrained=True)
        # 冻结部分骨干层，仅训练高层
        for name, param in self.backbone.named_parameters():
            if "layer1" in name or "layer2" in name:
                param.requires_grad = False
        # Anchor尺寸缩小，适配小目标
        self.rpn = RegionProposalNetwork(anchor_sizes=[16, 32, 64])
        # 数据增强：旋转、缩放、光照扰动
        self.aug = MicrobialAugmentation()
        self.head = DetectionHead(num_classes=12)
    def forward(self, images):
        images = self.aug(images)
        features = self.backbone(images)
        proposals = self.rpn(features)
        return self.head(features, proposals)

class LargeDatasetTrainer:

class OpenCVDetector:
    """OpenCV检测器"""
    def detect_petri_dish(self, image):
        # 霍夫圆检测培养皿
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
        blurred = cv2.GaussianBlur(gray, (9, 9), 2)
        circles = cv2.HoughCircles(
            blurred, cv2.HOUGH_GRADIENT,
            dp=1, minDist=400, param1=50, param2=35,
            minRadius=100, maxRadius=300
        )
        return self.validate_circles(circles)
    def detect_inhibition_zones(self, image, substances):
        # 多策略抑菌圈检测
        zones = []
        for substance in substances:
            roi = self.extract_roi(image, substance)
            # 滤纸片：亮度阈值+标准差验证
            if substance.type == "filter_paper":
                mask = self.create_mask(roi)
                mean = np.mean(roi[mask])
                std = np.std(roi[mask])
                if mean > 120 and std < 25:
                    zones.append(self.fit_circle(roi))
            # 透明圈：反色+低阈值+圆度判别
            elif substance.type == "hole":
                inv = cv2.bitwise_not(roi)
                edges = cv2.Canny(inv, 40, 120)
                contours = cv2.findContours(edges, ...)
                for cnt in contours:
                    if self.is_circular(cnt):
                        zones.append(self.fit_circle(cnt))
        return zones
工程细节：
  • 抑菌圈检测支持滤纸片与透明圈两种类型，参数自适应，亮度/标准差/圆度多重验证。
  • 小数据集模型采用骨干冻结、anchor缩放、数据增强等策略提升泛化。
  • 大数据集模型支持FP16混合精度与分布式训练，提升效率与精度。
