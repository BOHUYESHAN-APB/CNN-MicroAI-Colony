OpenCV传统视觉系统

基于OpenCV的计算机视觉算法，实现快速、轻量级的微生物菌落检测，适用于资源受限环境。

Hough圆检测

基于霍夫变换的圆检测算法，通过参数空间投票机制检测图像中的圆形目标。特别适合检测培养皿和菌落等圆形物体。

算法原理

  • 边缘检测：使用Canny算子检测图像边缘
  • 梯度计算：计算边缘点的梯度方向
  • 参数空间投票：在三维参数空间(a,b,r)中投票
  • 峰值检测：找到参数空间中的局部最大值
  • 圆心定位：根据峰值位置确定圆心和半径

import cv2 import numpy as np from typing import List, Tuple, Dict class ColonyDetector: """微生物菌落检测器""" def __init__(self): self.min_area = 50 self.max_area = 5000 self.min_circularity = 0.7 def preprocess_image(self, image: np.ndarray) -> np.ndarray: """图像预处理""" # 转换为灰度图 gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY) # CLAHE对比度增强 clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8,8)) enhanced = clahe.apply(gray) # 高斯滤波去噪 blurred = cv2.GaussianBlur(enhanced, (5, 5), 0) return blurred def detect_circles(self, image: np.ndarray) -> List[Tuple[int, int, int]]: """Hough圆检测""" circles = cv2.HoughCircles( image, cv2.HOUGH_GRADIENT, dp=1, minDist=20, param1=50, param2=30, minRadius=5, maxRadius=100 ) if circles is not None: circles = np.round(circles[0, :]).astype("int") return [(x, y, r) for x, y, r in circles] return [] def detect_contours(self, image: np.ndarray) -> List[Dict]: """轮廓检测""" # Canny边缘检测 edges = cv2.Canny(image, 50, 150) # 形态学操作 kernel = np.ones((3, 3), np.uint8) edges = cv2.morphologyEx(edges, cv2.MORPH_CLOSE, kernel) # 查找轮廓 contours, _ = cv2.findContours( edges, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE ) results = [] for contour in contours: area = cv2.contourArea(contour) if self.min_area <= area <= self.max_area: # 计算圆形度 perimeter = cv2.arcLength(contour, True) if perimeter > 0: circularity = 4 * np.pi * area / (perimeter * perimeter) if circularity >= self.min_circularity: # 计算中心点 M = cv2.moments(contour) if M["m00"] != 0: cx = int(M["m10"] / M["m00"]) cy = int(M["m01"] / M["m00"]) # 计算等效圆半径 radius = int(np.sqrt(area / np.pi)) results.append({ 'center': (cx, cy), 'radius': radius, 'area': area, 'circularity': circularity, 'contour': contour }) return results def analyze_colonies(self, image_path: str) -> Dict: """完整菌落分析""" # 读取图像 image = cv2.imread(image_path) if image is None: raise ValueError("无法读取图像") # 预处理 processed = self.preprocess_image(image) # 圆检测 circles = self.detect_circles(processed) # 轮廓检测 contours = self.detect_contours(processed) # 合并结果 all_colonies = [] # 添加圆检测结果 for x, y, r in circles: all_colonies.append({ 'type': 'circle', 'center': (x, y), 'radius': r, 'method': 'hough' }) # 添加轮廓检测结果 for contour in contours: all_colonies.append({ 'type': 'contour', 'center': contour['center'], 'radius': contour['radius'], 'area': contour['area'], 'circularity': contour['circularity'], 'method': 'contour' }) # 去重和筛选 filtered_colonies = self.filter_overlapping(all_colonies) # 统计分析 stats = { 'total_count': len(filtered_colonies), 'average_area': np.mean([c.get('area', np.pi * c['radius']**2) for c in filtered_colonies]) if filtered_colonies else 0, 'min_area': min([c.get('area', np.pi * c['radius']**2) for c in filtered_colonies]) if filtered_colonies else 0, 'max_area': max([c.get('area', np.pi * c['radius']**2) for c in filtered_colonies]) if filtered_colonies else 0, 'colonies': filtered_colonies } return stats def filter_overlapping(self, colonies: List[Dict]) -> List[Dict]: """过滤重叠检测结果""" if not colonies: return [] # 按面积排序 colonies = sorted(colonies, key=lambda x: x.get('area', np.pi * x['radius']**2), reverse=True) filtered = [] for colony in colonies: is_overlapping = False cx, cy = colony['center'] for existing in filtered: ex, ey = existing['center'] distance = np.sqrt((cx - ex)**2 + (cy - ey)**2) # 如果中心点距离小于较小半径的0.5倍，认为是重叠 min_radius = min(colony['radius'], existing['radius']) if distance < min_radius * 0.5: is_overlapping = True break if not is_overlapping: filtered.append(colony) return filtered # 使用示例 if __name__ == "__main__": detector = ColonyDetector() # 分析图像 results = detector.analyze_colonies("sample.jpg") print(f"检测到菌落数量: {results['total_count']}") print(f"平均面积: {results['average_area']:.2f} 像素") print(f"面积范围: {results['min_area']:.2f} - {results['max_area']:.2f} 像素")
